  • ✓ Pass rateTrail returns 'moderate' and 'easy' at the right limits: (6, 400) and (10, 2000) are moderate, (5, 800) is easy
  • ✓ Pass canJoin: 18 and older always, 14 to 17 only with a waiver, under 14 never

  • ✓ Pass Your code uses else if, a logical operator (&& or ||) and a switch

// Strictest rule first: only the first true branch runs
function rateTrail(miles, climb) {
  if (miles > 10 || climb > 2000) {
    return 'hard';
  } else if (miles > 5 || climb > 800) {
    return 'moderate';
  } else {
    return 'easy';
  }
}

// Two ways in: old enough on their own, OR 14+ with a signed waiver
function canJoin(age, hasWaiver) {
  return age >= 18 || (age >= 14 && hasWaiver === true);
}

// return leaves the function, so no break is needed after each case
function gearNote(forecast) {
  switch (forecast) {
    case 'sun':
      return 'Bring sunscreen';
    case 'rain':
      return 'Bring a rain jacket';
    case 'snow':
      return 'Trip cancelled';
    default:
      return 'Check the forecast again';
  }
}

console.log(canJoin(15, true), canJoin(15, false)); // true false

Challenge 8.7: Functions, scope and a bottle that remembers

6 of 6 checks pass

  • ✓ Pass minutesPerMile is a function declaration: (6, 3) gives 30 and (4, 1.5) gives 23
  • ✓ Pass toKm is an arrow function: toKm(10) is 16.1 and toKm(26.2) is 42.2
  • ✓ Pass welcome('Ana') is 'Welcome, Ana!' and welcome() is 'Welcome, hiker!' (a default parameter)
  • ✓ Pass trailPoints(6, 1500) is 11 and trailPoints(6, 200) is 6: no more ReferenceError
  • ✓ Pass Each bottle remembers its own water over several drinks: drink(300) twice leaves 400, it never goes below 0, and left() reports it
  • ✓ Pass The water is private: a bottle object holds only its two methods

JavaScript

// Function declaration
function minutesPerMile(miles, hours) {
  return Math.round((hours * 60) / miles);
}

// Arrow function: no braces, so the expression is returned automatically
const toKm = (miles) => Math.round(miles * 1.609 * 10) / 10;

// Default parameter: used when the argument is missing
function welcome(person = 'hiker') {
  return `Welcome, ${person}!`;
}

// Declare bonus before the if, so it still exists at the return
function trailPoints(miles, climb) {
  let bonus = 0;
  if (climb > 1000) {
    bonus = 5;
  }
  return miles + bonus;
}

// Closure: water lives inside makeBottle; only drink() and left() can reach it
function makeBottle(capacity) {
  let water = capacity;
  return {
    drink(ml) {
      water = Math.max(0, water - ml);
      return water;
    },
    left() {
      return water;
    },
  };
}

console.log(minutesPerMile(6, 3), toKm(10)); // 30 16.1
console.log(welcome('Ana'), welcome()); // Welcome, Ana! Welcome, hiker!
console.log(trailPoints(6, 1500), trailPoints(6, 200)); // 11 6
const mine = makeBottle(1000);
const yours = makeBottle(500);
console.log(mine.drink(300), yours.left()); // 700 500

// One function does the filtering, so every event path stays in step
function applySearch() {
  const text = search.value.trim().toLowerCase();
  let visible = 0;
  items.forEach((item) => {
    const match = item.textContent.toLowerCase().includes(text);
    item.hidden = !match;

    left.classList.remove('over');
  } else {
    left.textContent = `${-remaining} characters too many`;
    left.classList.add('over');
  }

// Shows one field's error: text for everyone, aria-invalid for assistive technology
function showError(field, message) {

  field.setAttribute('aria-invalid', 'true');
}

// Clears one field's error once its value is fine
function clearError(field) {
  document.getElementById(field.id + '-error').textContent = '';
  field.setAttribute('aria-invalid', 'false');
}

const controls = ['fullname', 'email', 'group', 'pace', 'waiver'].map((n) => form.elements[n]);

const messages = {
  fullname: 'Please enter your full name.',
  email: 'Enter an email address like name@example.com.',
  group: 'Enter a group size from 1 to 6.',
  pace: 'Choose a pace from the list.',
  waiver: 'Tick the box to confirm you have read the safety waiver.',
};

// Checks one control, shows or clears its error, and says whether it passed
function checkControl(control) {
  if (control.checkValidity()) {
    clearError(control);
    return true;
  }
  showError(control, messages[control.id]);
  return false;
}

form.addEventListener('submit', (event) => {

// The club's autumn trips. Your code draws the whole list from this array.
const trips = [
  { id: 'ridge', title: 'Autumn Ridge Walk', miles: 6, spots: 2 },
  { id: 'lake', title: 'Lake Loop', miles: 3.5, spots: 0 },
  { id: 'falls', title: 'Cedar Falls', miles: 9, spots: 5 },
  { id: 'summit', title: 'Summit Sunrise', miles: 12, spots: 1 }
];
const joined = []; // ids of joined trips, in the order joined

const list = document.querySelector('#trips');
const search = document.querySelector('#search');
const summary = document.querySelector('#summary');
const mine = document.querySelector('#mine');

// The ONE place that draws the page from the data
function render() {
  const text = search.value.trim().toLowerCase();
  const matches = trips.filter((trip) => trip.title.toLowerCase().includes(text));

  list.textContent = ''; // empty first, or every redraw adds more cards

  matches.forEach((trip) => {
    const li = document.createElement('li');
    li.classList.add('trip');

    const h2 = document.createElement('h2');
    h2.textContent = trip.title;

    const p = document.createElement('p');
    p.classList.add('spots');
    const word = trip.spots === 1 ? 'spot' : 'spots';
    p.textContent = `${trip.miles} miles, ${trip.spots} ${word} left`;

    const button = document.createElement('button');
    button.type = 'button';
    button.classList.add('join');
    button.dataset.id = trip.id;

    if (trip.spots === 0) {
      li.classList.add('full');
    }
    if (joined.includes(trip.id)) {
      button.textContent = 'Joined';
      button.disabled = true;
    } else if (trip.spots === 0) {
      button.textContent = 'Full';
      button.disabled = true;
    } else {
      button.textContent = 'Join';
    }

    li.append(h2, p, button);
    list.append(li);
  });

  // spots across ALL trips, even ones the search hides
  const totalSpots = trips.reduce((sum, trip) => sum + trip.spots, 0);
  summary.textContent = `Showing ${matches.length} of ${trips.length} trips, ${totalSpots} spots left`;

  if (joined.length === 0) {
    mine.textContent = 'You have not joined a trip yet.';
  } else {
    const titles = joined.map((id) => trips.find((trip) => trip.id === id).title);
    mine.textContent = `You joined: ${titles.join(', ')}`;
  }
}

  const button = event.target.closest('.join');

  if (!trip || trip.spots === 0 || joined.includes(trip.id)) return;
  trip.spots -= 1;
  joined.push(trip.id);
  render();
});

In Challenge 8.1, the script in the <head> found nothing until you changed it. Explain why, and describe two different ways to make sure a script runs only after the elements it needs exist.

It found nothing because the browser reads the HTML from top to bottom and runs a <script> as soon as it reaches it. A script in the <head> runs before the browser has reached the <body>, so the elements it looks for don’t exist yet. You can add defer to the script tag or move the script to the end of the body element.

A text box gives your script the value '5'. Explain why '5' + 1 and Number('5') + 1 give different results, and why === is a safer way than == to compare values like these.

They give different results because text boxes always return a string. When either side of + is a string, JavaScript joins the values as text instead of adding them. Number('5') + 1 gives 6, because Number() turns the string into a real number first. === is safer than == because == quietly converts types before comparing, so '5' == 5 and even '' == 0 are true. === checks type and value with no conversion, so '5' === 5 is false, and you have to convert the input yourself, which keeps the code clear and avoids hidden bugs.

Explain how makeBottle keeps each bottle's water separate and private. Where does the water variable live, and why does it still exist after makeBottle has finished running?

Water is a local variable declared with let inside makeBottle, and it starts at capacity. Each call creates a new object with its own water, so makeBottle 1000 and makeBottle 500 give mine and yours two completely separate variables. The variable is private because nothing outside the function can reach it. The variable water still exists after makeBottle returns because of a closure. JavaScript keeps that water alive for as long as the bottle object exists.

Explain why you should use textContent rather than innerHTML to show text that a visitor typed. What could go wrong with innerHTML?

You should use textContent because it shows a visitor’s input as plain text, so <b>hi</b> appears exactly as typed. Now innerHTML turns the input into real HTML, which lets an attacker inject code. This is called cross-site scripting (XSS). Real attacker could use the same trick to steal cookies or change the page for other visitors. textContent keeps user input as harmless text that can’t run.

Explain event delegation in your own words. Why does one listener on #filters hear clicks on every button, and why do you need closest() instead of using event.target as it is?

Event delegation means putting one listener on a parent instead of one on each child. Clicks bubble up from the clicked element through its ancestors, so every click on a button inside #filters reaches #filters, and one listener hears them all, even buttons added later. event.target is the exact element clicked, which might be a <span> or icon inside the button, or the gap between buttons. event.target.closest('button') walks up to the nearest button, or returns null if the click wasn’t on one, so you always get the button itself.

Your script checks the sign-up form before anything is sent. Explain why the club's server must still check the same data, and how aria-invalid and a text message under each field help a visitor who cannot see a red border.

The club's server must still check the same data because the browser check is only a convenience for the visitor, and anyone can turn off JavaScript, edit the page in DevTools or send data straight to the server without the form. The server is the only check visitors can’t skip, so it must validate the data again before saving it. For visitors who can’t see a red border, aria-invalid="true" tells screen readers the field has an error. A text message under the field, linked with aria-describedby, explains what’s wrong, such as “Enter a valid email.” This works for screen reader users, people with color blindness and everyone else.
